```python
#!/usr/bin/env python3
"""
A simple, reusable web scraper.

Features:
- Configurable target URLs (single page or list of pages)
- Polite rate‑limiting (default 1 request per second)
- Automatic retry with exponential back‑off
- HTML parsing with BeautifulSoup (lxml parser)
- Easy extraction via CSS selectors or custom functions
- Optional CSV output
- Logging & progress bar

Author: ChatGPT (OpenAI)
"""

import csv
import time
import random
import logging
from pathlib import Path
from typing import List, Callable, Dict, Any, Iterable, Optional

import requests
from bs4 import BeautifulSoup
from tqdm import tqdm

# ----------------------------------------------------------------------
# Configuration
# ----------------------------------------------------------------------
USER_AGENT = (
    "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
    "AppleWebKit/537.36 (KHTML, like Gecko) "
    "Chrome/124.0.0.0 Safari/537.36"
)

HEADERS = {"User-Agent": USER_AGENT}
TIMEOUT = 15               # seconds
MAX_RETRIES = 3
BACKOFF_FACTOR = 1.5       # multiplier for exponential back‑off
RATE_LIMIT = 1.0           # seconds between requests (politeness)
OUTPUT_CSV = "scraped_data.csv"

# ----------------------------------------------------------------------
# Logging setup
# ----------------------------------------------------------------------
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    handlers=[logging.StreamHandler()],
)
log = logging.getLogger(__name__)

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def fetch(url: str) -> Optional[requests.Response]:
    """Fetch a URL with retries and exponential back‑off."""
    delay = RATE_LIMIT
    for attempt in range(1, MAX_RETRIES + 1):
        try:
            log.debug(f"Fetching {url} (attempt {attempt})")
            response = requests.get(url, headers=HEADERS, timeout=TIMEOUT)
            response.raise_for_status()
            return response
        except (requests.RequestException, requests.HTTPError) as exc:
            log.warning(f"Error fetching {url}: {exc}")
            if attempt == MAX_RETRIES:
                log.error(f"Giving up on {url}")
                return None
            # exponential back‑off + jitter
            sleep_time = delay * (BACKOFF_FACTOR ** (attempt - 1))
            sleep_time += random.uniform(0, 0.5)
            log.info(f"Retrying in {sleep_time:.2f}s …")
            time.sleep(sleep_time)
    return None


def parse_html(html: str) -> BeautifulSoup:
    """Parse raw HTML into a BeautifulSoup object."""
    return BeautifulSoup(html, "lxml")


def extract_items(
    soup: BeautifulSoup,
    selector: str,
    extractor: Callable[[BeautifulSoup], Dict[str, Any]],
) -> List[Dict[str, Any]]:
    """
    Find elements matching `selector` and run `extractor` on each.

    Parameters
    ----------
    soup : BeautifulSoup
        Parsed page.
    selector : str
        CSS selector that isolates each “record” (e.g., "article.post").
    extractor : Callable
        Function that receives a BeautifulSoup element representing a single
        record and returns a dict of the fields you want.

    Returns
    -------
    List[dict]
        Extracted rows.
    """
    items = []
    for element in soup.select(selector):
        try:
            data = extractor(element)
            items.append(data)
        except Exception as exc:
            log.debug(f"Skipping element due to extraction error: {exc}")
    return items


def write_csv(path: Path, rows: Iterable[Dict[str, Any]], fieldnames: List[str]) -> None:
    """Write rows to a CSV file (creates/overwrites)."""
    with path.open("w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        for row in rows:
            writer.writerow(row)


# ----------------------------------------------------------------------
# Example scraper implementation
# ----------------------------------------------------------------------
def example_extractor(article_tag: BeautifulSoup) -> Dict[str, Any]:
    """
    Custom extractor for a generic blog‑style page.
    Adjust the CSS selectors to match the site you target.
    """
    title_tag = article_tag.select_one("h2.entry-title a")
    date_tag = article_tag.select_one("time.published")
    summary_tag = article_tag.select_one("div.entry-summary")

    return {
        "title": title_tag.get_text(strip=True) if title_tag else "",
        "url": title_tag["href"] if title_tag and title_tag.has_attr("href") else "",
        "date": date_tag["datetime"] if date_tag and date_tag.has_attr("datetime") else "",
        "summary": summary_tag.get_text(strip=True) if summary_tag else "",
    }


def scrape_pages(urls: List[str]) -> List[Dict[str, Any]]:
    """
    Main driver: fetch each URL, parse, extract, and aggregate results.
    """
    all_rows: List[Dict[str, Any]] = []

    for url in tqdm(urls, desc="Pages", unit="page"):
        resp = fetch(url)
        if resp is None:
            continue

        soup = parse_html(resp.text)

        # Adjust selector & extractor for your target site
        rows = extract_items(
            soup,
            selector="article.post",   # <-- change this to match the container
            extractor=example_extractor,
        )
        all_rows.extend(rows)

        # Polite pause (already handled by fetch's RATE_LIMIT, but you can add extra)
        time.sleep(RATE_LIMIT)

    return all_rows


# ----------------------------------------------------------------------
# Entry point (run as script)
# ----------------------------------------------------------------------
if __name__ == "__main__":
    # ------------------------------------------------------------------
    # 1️⃣ Define the list of pages you want to scrape.
    # ------------------------------------------------------------------
    # Example: scrape the first 5 pages of a paginated blog.
    base = "https://example-blog.com/page/{}"
    pages_to_scrape = [base.format(i) for i in range(1, 6)]

    # ------------------------------------------------------------------
    # 2️⃣ Run the scraper.
    # ------------------------------------------------------------------
    log.info("Starting scraper …")
    results = scrape_pages(pages_to_scrape)

    if not results:
        log.warning("No data extracted – double‑check your selectors/extractor.")
    else:
        # ------------------------------------------------------------------
        # 3️⃣ Write to CSV (optional but handy for analysis)
        # ------------------------------------------------------------------
        field_names = sorted(results[0].keys())
        write_csv(Path(OUTPUT_CSV), results, field_names)
        log.info(f"✅ Finished! {len(results)} records saved to {OUTPUT_CSV}")

```

## How to adapt it to **your** target site

1. **Identify the page(s)** you need to scrape.  
   - If the site uses pagination, generate the list of URLs (see the `pages_to_scrape` example).  
   - For a single page, just put that URL in a list.

2. **Find the HTML container** that holds each record you care about.  
   - Open the page in Chrome/Firefox, right‑click → *Inspect*.  
   - Look for a repeating element (e.g., `
`, `
`, etc.).  
   - Use that as the `selector` argument in `extract_items`.

3. **Write an extractor function** (`example_extractor` in the script).  
   - Inside the function, use `element.select_one("css")` or `element.find(...)` to pull out title, link, price, etc.  
   - Return a dictionary where keys become CSV column names.

4. **Adjust optional settings**:  
   - `RATE_LIMIT` → how many seconds to wait between requests (be courteous).  
   - `MAX_RETRIES` / `BACKOFF_FACTOR` → how aggressively to retry failed requests.  
   - `HEADERS` → you can add cookies or other headers if the site requires them.

5. **Run**:

```bash
python scraper.py
```

You’ll get a `scraped_data.csv` file with the extracted rows.
